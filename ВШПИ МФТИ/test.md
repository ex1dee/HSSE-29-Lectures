\begin{cases}
x(t) = t\cos t + \sin t \\
y(t) = \cos t + 2t \\
z(t) = t
\end{cases}
$$

$$r(t) = (t\cos t + \sin t, \cos t + 2t, t)$$
$$
r'(t) = (\cos t - t\sin t + \cos t, 2 - \sin t, 1) = (2\cos t - t\sin t, 2 - \sin t, 1) = (2, 2, 1)
$$
$$
r''(t) = (-2\sin t - \sin t - t \cos t, 2 - \cos t, 0) = (-1, 1, 0)
$$
$$
r'(t) \times r''(t) = (-1, 1, 4), |r'(t) \times r''(t)| = \sqrt{ 18 } = 3\sqrt{ 2 }
$$
